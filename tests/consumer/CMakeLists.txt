cmake_minimum_required(VERSION 3.20)
project(TeklaConsumer LANGUAGES CXX)
find_package(TeklaFormats 0.3 CONFIG REQUIRED)
find_package(TeklaDB1 0.3 CONFIG REQUIRED)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE TeklaFormats::core TeklaDB1::core)
enable_testing()
add_executable(shapes_consumer ../shapes.cpp)
target_link_libraries(shapes_consumer PRIVATE TeklaFormats::core ZLIB::ZLIB)
add_test(NAME installed-shapes-consumer COMMAND shapes_consumer "${CMAKE_CURRENT_BINARY_DIR}/test-data" shapes_catalog)
add_test(NAME installed-shapes-gzip-consumer COMMAND shapes_consumer "${CMAKE_CURRENT_BINARY_DIR}/test-data" shapes_gzip)
add_executable(bolt844_consumer ../regression.cpp)
target_link_libraries(bolt844_consumer PRIVATE TeklaFormats::core ZLIB::ZLIB)
add_test(NAME installed-bolt844-consumer COMMAND bolt844_consumer "${CMAKE_CURRENT_BINARY_DIR}/test-data" bolt844_valid)
add_test(NAME installed-bolt844-library-consumer COMMAND bolt844_consumer "${CMAKE_CURRENT_BINARY_DIR}/test-data" bolt844_library_multi)
add_executable(guid_mapping_consumer ../guid_mappings.cpp)
target_link_libraries(guid_mapping_consumer PRIVATE TeklaFormats::core)
add_test(NAME installed-guid-mapping-consumer COMMAND guid_mapping_consumer "${CMAKE_CURRENT_BINARY_DIR}/test-data" mapper_text)
add_test(NAME installed-guid-mapping-targets-consumer COMMAND guid_mapping_consumer "${CMAKE_CURRENT_BINARY_DIR}/test-data" mapper_targets)
add_executable(metadata_consumer ../metadata.cpp)
target_link_libraries(metadata_consumer PRIVATE TeklaFormats::core)
add_test(NAME installed-metadata-consumer COMMAND metadata_consumer "${CMAKE_CURRENT_BINARY_DIR}/test-data" metadata_scope)
add_test(NAME installed-metadata-entities-consumer COMMAND metadata_consumer "${CMAKE_CURRENT_BINARY_DIR}/test-data" metadata_entities)
add_test(NAME installed-consumer COMMAND consumer)
add_executable(drawing_consumer ../related.cpp)
target_link_libraries(drawing_consumer PRIVATE TeklaFormats::core)
add_test(NAME installed-drawing-dimensions-consumer COMMAND drawing_consumer "${CMAKE_CURRENT_BINARY_DIR}/test-data" dimensions_valid)
add_test(NAME installed-drawing844-consumer COMMAND drawing_consumer "${CMAKE_CURRENT_BINARY_DIR}/test-data" drawing844_valid)
add_test(NAME installed-drawing730-dimensions-consumer COMMAND drawing_consumer "${CMAKE_CURRENT_BINARY_DIR}/test-data" dimensions730_valid)
if(TARGET TeklaFormats::occt)
    add_executable(geometry_consumer ../geometry.cpp)
    target_link_libraries(geometry_consumer PRIVATE TeklaFormats::occt TeklaDB1::occt)
    add_test(NAME installed-geometry-consumer COMMAND geometry_consumer)
    add_test(NAME installed-nominal-geometry-consumer COMMAND geometry_consumer nominal)
    add_test(NAME installed-polybeam-miter-consumer COMMAND geometry_consumer miter)
    add_test(NAME installed-cambering-consumer COMMAND geometry_consumer camber)
    add_test(NAME installed-ellipse-consumer COMMAND geometry_consumer ellipse)
    add_test(NAME installed-bolt844-geometry-consumer COMMAND geometry_consumer bolt844)
endif()
