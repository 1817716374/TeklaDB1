cmake_minimum_required(VERSION 3.20)
project(TeklaConsumer LANGUAGES CXX)
find_package(TeklaFormats 0.3 CONFIG REQUIRED)
find_package(TeklaDB1 0.3 CONFIG REQUIRED)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE TeklaFormats::core TeklaDB1::core)
enable_testing()
add_test(NAME installed-consumer COMMAND consumer)
if(TARGET TeklaFormats::occt)
    add_executable(geometry_consumer ../geometry.cpp)
    target_link_libraries(geometry_consumer PRIVATE TeklaFormats::occt TeklaDB1::occt)
    add_test(NAME installed-geometry-consumer COMMAND geometry_consumer)
    add_test(NAME installed-nominal-geometry-consumer COMMAND geometry_consumer nominal)
    add_test(NAME installed-polybeam-miter-consumer COMMAND geometry_consumer miter)
    add_test(NAME installed-cambering-consumer COMMAND geometry_consumer camber)
endif()
